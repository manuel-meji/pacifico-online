-- 2026_10_21_100000_create_comercios_table.php
create table comercios (
    id_comercio bigserial primary key,
    cedula_juridica varchar(12) not null,
    nombre_comercial varchar(150) not null,
    razon_social varchar(150) not null,
    direccion_fiscal varchar(255) not null,
    inscrito_como_contribuyente boolean not null default false,
    estado varchar(15) not null default 'Pendiente',
    created_at timestamp(0) without time zone null,
    updated_at timestamp(0) without time zone null
);
alter table comercios add constraint uq_comercios_cedula_juridica unique (cedula_juridica);
alter table comercios add constraint chk_comercios_estado check (estado IN ('Pendiente', 'Activo', 'Rechazado', 'Suspendido'));  -- Estados del comercio (glosario)
alter table comercios add constraint chk_comercios_cedula_formato check (cedula_juridica ~ '^[0-9]-[0-9]{3}-[0-9]{6}$');  -- Formato de cédula jurídica

alter table membresias_comercio add constraint chk_membresia_comision_preferencial check (comision_preferencial IS NULL OR (comision_preferencial >= 0 AND comision_preferencial <= 100));  -- RN-008
